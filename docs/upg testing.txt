interface AnalysisRule {
  id: string;
  category: 'risk' | 'performance' | 'execution' | 'universe' | 'market_specific';
  condition: RuleCondition;
  recommendations: RecommendationTemplate[];
  priority: 'critical' | 'high' | 'medium' | 'low';
  description: string;
}

interface RuleCondition {
  field: keyof BacktestMetrics;
  operator: '>' | '<' | '>=' | '<=' | '==';
  value: number | string;
  and?: RuleCondition[];
  or?: RuleCondition[];
}

interface RecommendationTemplate {
  field: string;          // Поле для изменения
  operation: 'increase' | 'decrease' | 'set' | 'toggle' | 'suggest';
  value: number | string | ((metrics: BacktestMetrics) => number | string);
  reasoning: string;
  expectedImpact: string;
  riskLevel: 'low' | 'medium' | 'high';
  confidenceBonus?: number; // Дополнительный бонус к confidence
}

3. АЛГОРИТМ ОПТИМИЗАЦИИ
3.1. Стратегия оптимизации
Используем комбинированный подход:

Грубый перебор (Grid Search) — быстрый прогон по широкой сетке для выявления перспективных зон.

Уточняющий перебор (Fine-Tuning) — детальный перебор в найденных зонах.

Итеративное улучшение — повторение шагов 1-2 с сужением диапазонов.

  config: TradingRobotConfig;
  metrics: BacktestMetrics;
  score: number;
  rank: number;
}

async function runOptimization(
  baseConfig: TradingRobotConfig,
  optimizationGoal: OptimizationGoal,
  maxIterations: number = 50
): Promise<OptimizationResult[]> {
  
  // Шаг 1: Определить параметры для оптимизации
  const params = identifyOptimizableParams(baseConfig);
  
  // Шаг 2: Сгенерировать сетку (Grid)
  const grid = generateGrid(params, optimizationGoal);
  
  // Шаг 3: Запустить бэктесты (параллельно)
  const results = await runBacktestsInParallel(grid, maxIterations);
  
  // Шаг 4: Оценить результаты
  const scoredResults = results.map(r => ({
    ...r,
    score: calculateScore(r.metrics, optimizationGoal)
  }));
  
  // Шаг 5: Отсортировать по убыванию score
  scoredResults.sort((a, b) => b.score - a.score);
  
  return scoredResults;
}
3.3. Генерация сетки значений

function generateGrid(params: OptimizableParam[], goal: OptimizationGoal): TradingRobotConfig[] {
  // Для каждого параметра генерируем список значений
  const paramValues: Record<string, any[]> = {};
  
  for (const param of params) {
    const values: any[] = [];
    
    // Если параметр уже имеет "горячую" зону (из предыдущих оптимизаций)
    if (param.hotZone) {
      // Густая сетка в горячей зоне
      for (let v = param.hotZone.start; v <= param.hotZone.end; v += param.hotZone.step) {
        values.push(v);
      }
    } else {
      // Редкая сетка по всему диапазону (грубый перебор)
      for (let v = param.min; v <= param.max; v += param.step * 2) {
        values.push(v);
      }
    }
    
    paramValues[param.field] = values;
  }
  
  // Генерируем комбинации (Cartesian product)
  const combinations = cartesianProduct(paramValues);
  
  // Ограничиваем количество комбинаций (чтобы не упала производительность)
  const maxCombinations = goal === 'speed' ? 20 : 50;
  const sampledCombinations = sampleCombinations(combinations, maxCombinations);
  
  return sampledCombinations.map(combo => applyParamsToConfig(baseConfig, combo));
}
3.4. Оценка результата (Scoring)
typescript
function calculateScore(metrics: BacktestMetrics, goal: OptimizationGoal): number {
  let score = 0;
  
  switch (goal.type) {
    case 'max_return':
      // Максимальная доходность
      score = metrics.totalReturn * 0.6 
            - metrics.maxDrawdown * 0.25 
            + metrics.sharpeRatio * 0.15;
      break;
      
    case 'min_drawdown':
      // Минимальная просадка
      score = -metrics.maxDrawdown * 0.5 
            + metrics.totalReturn * 0.3 
            + metrics.sharpeRatio * 0.2;
      break;
      
    case 'max_sharpe':
      // Максимальный Sharpe ratio
      score = metrics.sharpeRatio * 0.6 
            - metrics.maxDrawdown * 0.2 
            + metrics.totalReturn * 0.2;
      break;
      
    case 'balanced':
      // Сбалансированный (по умолчанию)
      score = metrics.totalReturn * 0.35 
            - metrics.maxDrawdown * 0.30 
            + metrics.sharpeRatio * 0.20 
            + metrics.winRate * 0.15;
      break;
      
    case 'custom':
      // Пользовательский (с весами)
      score = metrics.totalReturn * goal.weights.return
            - metrics.maxDrawdown * goal.weights.drawdown
            + metrics.sharpeRatio * goal.weights.sharpe
            + metrics.winRate * goal.weights.winRate
            - metrics.totalTrades * goal.weights.trades;
      break;
  }
  
  // Штраф за слишком малое количество сделок
  if (metrics.totalTrades < 10) {
    score *= 0.5;
  }
  
  // Штраф за слишком высокую просадку
  if (metrics.maxDrawdown > 30) {
    score *= 0.7;
  }
  
  // Бонус за стабильность (низкий standard deviation)
  if (metrics.stdDev < 0.2) {
    score *= 1.1;
  }
  
  return score;
}
3.5. Итеративное улучшение (Adaptive Optimization)

async function adaptiveOptimization(
  baseConfig: TradingRobotConfig,
  goal: OptimizationGoal,
  maxIterations: number = 50
): Promise<OptimizationResult[]> {
  
  let currentBest: OptimizationResult | null = null;
  let allResults: OptimizationResult[] = [];
  let searchSpace = getInitialSearchSpace(baseConfig);
  
  for (let iteration = 0; iteration < 5; iteration++) {
    // 1. Сгенерировать сетку в текущем поисковом пространстве
    const grid = generateGrid(searchSpace, goal);
    
    // 2. Запустить бэктесты
    const results = await runBacktestsInParallel(grid, maxIterations);
    allResults = [...allResults, ...results];
    
    // 3. Найти лучший результат
    const best = results.reduce((a, b) => a.score > b.score ? a : b);
    
    // 4. Обновить поисковое пространство (сузить вокруг лучшего)
    searchSpace = narrowSearchSpace(searchSpace, best.config);
    
    // 5. Сохранить лучший результат
    if (!currentBest || best.score > currentBest.score) {
      currentBest = best;
    }
    
    // 6. Если улучшение меньше 1% — остановиться
    if (iteration > 0 && (currentBest.score - prevBest.score) / prevBest.score < 0.01) {
      break;
    }
    
    prevBest = currentBest;
  }
  
  // Сортировка всех результатов
  allResults.sort((a, b) => b.score - a.score);
  
  return allResults;
}
3.6. Сужение поискового пространства
typescript
function narrowSearchSpace(
  currentSpace: SearchSpace,
  bestConfig: TradingRobotConfig
): SearchSpace {
  const newSpace: SearchSpace = {};
  
  for (const [param, range] of Object.entries(currentSpace)) {
    const bestValue = getParamValue(bestConfig, param);
    const halfWidth = (range.max - range.min) * 0.3; // Сужаем на 30%
    
    newSpace[param] = {
      min: Math.max(range.min, bestValue - halfWidth),
      max: Math.min(range.max, bestValue + halfWidth),
      step: range.step * 0.5, // Уменьшаем шаг
    };
  }
  
  return newSpace;
}

function checkOverfitting(results: OptimizationResult[]): boolean {
  // 1. Проверка на "слишком хороший" результат
  const best = results[0];
  const secondBest = results[1];
  
  // Если лучший результат сильно отличается от второго (> 20%)
  if (best.score > secondBest.score * 1.2) {
    return true; // Подозрение на переоптимизацию
  }
  
  // 2. Проверка на количество сделок
  if (best.metrics.totalTrades < 20) {
    return true; // Слишком мало сделок — результат нестабилен
  }
  
  // 3. Проверка на Sharpe ratio
  if (best.metrics.sharpeRatio > 2.5) {
    return true; // Слишком высокий Sharpe — подозрительно
  }
  
  return false;
}

function validateResult(metrics: BacktestMetrics): ValidationResult {
  const issues: string[] = [];
  
  // 1. Проверка на "слишком" высокую доходность
  if (metrics.totalReturn > 100) {
    issues.push('Доходность > 100% — возможна ошибка в данных или переоптимизация');
  }
  
  // 2. Проверка на "слишком" высокий Win Rate
  if (metrics.winRate > 80) {
    issues.push('Win Rate > 80% — результат нереалистичен');
  }
  
  // 3. Проверка на "слишком" высокий Sharpe
  if (metrics.sharpeRatio > 3.0) {
    issues.push('Sharpe ratio > 3.0 — результат нереалистичен');
  }
  
  // 4. Проверка на "слишком" низкую просадку при высокой доходности
  if (metrics.totalReturn > 30 && metrics.maxDrawdown < 5) {
    issues.push('Доходность > 30% при просадке < 5% — подозрительно');
  }
  
  return {
    isValid: issues.length === 0,
    issues
  };
}
